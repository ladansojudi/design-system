// ── Card wrapper ───────────────────────────────────────────────────────────

export interface ChartCardProps extends React.HTMLAttributes<HTMLDivElement> {
  title:        string;
  description?: string;
  viewAll?:     boolean;
  trailing?:    React.ReactNode;
  onViewAll?:   () => void;

export const ChartCard = React.forwardRef<HTMLDivElement, ChartCardProps>(

    return (
      <div
        ref={ref}

      >
        <div className="flex items-center gap-3 px-4 sm:px-5 py-3 bg-s4e-neutral-grey-100 border-b border-s4e-neutral-divider-10 rounded-t-xl">
          <span className="text-[14px] font-semibold text-s4e-text-primary shrink-0">{title}</span>

            <span className="hidden sm:inline text-[12px] text-s4e-text-disabled truncate flex-1 min-w-0">{description}</span>

            <button

              View all

  },

// ── Donut ──────────────────────────────────────────────────────────────────

export type DonutSegment = { label: string; value: number; color: string };

export interface DonutChartProps extends React.SVGAttributes<SVGSVGElement> {
  data:    DonutSegment[];
  size?:   number;
  showTotal?: boolean;
}

export const DonutChart = React.forwardRef<SVGSVGElement, DonutChartProps>(
  ({ data, size = 180, showTotal = true, className, ...props }, ref) => {
    const total = data.reduce((s, d) => s + d.value, 0);
    const RADIUS = 70;
    const STROKE = 28;
    const PATH_LEN = 100;

    let offset = 0;
    const segments = data.map((d) => {
      const dash = (d.value / total) * PATH_LEN;
      const seg  = { ...d, dash, offset };
      offset += dash;
      return seg;
    });

    return (
      <svg
        ref={ref}
        width={size}
        height={size}
        viewBox="-90 -90 180 180"
        className={cn("shrink-0", className)}
        {...props}
      >
        <circle r={RADIUS} fill="none" stroke="var(--s4e-neutral-grey-100)" strokeWidth={STROKE} />
        {segments.map((s) => (
          <circle
            key={s.label}
            r={RADIUS}
            fill="none"
            stroke={s.color}
            strokeWidth={STROKE}
            pathLength={PATH_LEN}
            strokeDasharray={`${s.dash} ${PATH_LEN - s.dash}`}
            strokeDashoffset={-s.offset}
            transform="rotate(-90)"
          />
        ))}
        {showTotal && (
          <text x="0" y="6" textAnchor="middle" className="fill-s4e-text-primary" fontSize="22" fontWeight="600">
            {total.toLocaleString()}
          </text>
        )}
      </svg>
    );
  },
);
DonutChart.displayName = "DonutChart";

// ── Bar ────────────────────────────────────────────────────────────────────

export type BarDatum = { label: string; value: number };

export interface BarChartProps extends React.HTMLAttributes<HTMLDivElement> {
  data:      BarDatum[];
  max?:      number;
  ticks?:    number[];
  height?:   number;
  highlightLeader?: boolean;
}

export const BarChart = React.forwardRef<HTMLDivElement, BarChartProps>(
  (
    {
      data,
      max,
      ticks,
      height = 240,
      highlightLeader = true,
      className,
      ...props
    },
    ref,
  ) => {
    const computedMax  = max ?? Math.max(...data.map((d) => d.value));
    const computedTicks = ticks ?? [0, 10, 20, 30, 40, 50].filter((t) => t <= computedMax || t === 0);
    const finalTicks   = computedTicks.length > 1 ? computedTicks : [0, computedMax];

    return (
      <div ref={ref} className={cn("space-y-4", className)} {...props}>
        <div className="flex gap-3">
          {/* Y axis labels */}
          <div
            className="flex flex-col-reverse justify-between text-right pr-1 w-8"
            style={{ height }}
          >
            {finalTicks.map((t) => (
              <span key={t} className="text-[10px] text-s4e-text-disabled tabular-nums">{t}</span>
            ))}
          </div>

          {/* Chart area */}
          <div className="flex-1 relative">
            <div className="absolute inset-0 flex flex-col-reverse justify-between pointer-events-none">
              {finalTicks.map((t) => (
                <div key={t} className="border-t border-s4e-neutral-divider-10/70" />
              ))}
            </div>
            <div className="relative flex items-end gap-2 sm:gap-6" style={{ height }}>
              {data.map((b, i) => (
                <div key={b.label} className="flex-1 flex flex-col items-center justify-end gap-1.5 h-full">
                  <span className="text-[11px] font-semibold text-s4e-text-primary tabular-nums">{b.value}</span>
                  <div
                    className={cn(
                      "w-full max-w-[54px] rounded-t-md transition-all",
                      highlightLeader && i === 0
                        ? "bg-s4e-brand-primary-600"
                        : "bg-s4e-brand-primary-500",
                    )}
                    style={{ height: `${(b.value / computedMax) * 100}%` }}
                  />
                </div>
              ))}
            </div>
          </div>
        </div>

        {/* X axis labels */}
        <div className="flex gap-2 sm:gap-6 pl-11">
          {data.map((b) => (
            <div key={b.label} className="flex-1 text-center text-[11px] text-s4e-text-secondary">
              {b.label}
            </div>
          ))}
        </div>
      </div>
    );
  },
);
BarChart.displayName = "BarChart";

// ── Line / Area ────────────────────────────────────────────────────────────

export interface LineChartProps extends React.HTMLAttributes<HTMLDivElement> {
  linePath:    string;
  yLabels?:    string[];
  xLabels?:    string[];
  showArea?:   boolean;
  gradientId?: string;
}

export const LineChart = React.forwardRef<HTMLDivElement, LineChartProps>(
  (
    {
      linePath,
      yLabels = ["1500", "1000", "500", "0"],
      xLabels = [],
      showArea = true,
      gradientId,
      className,
      ...props
    },
    ref,
  ) => {
    const reactId = React.useId();
    const fillId  = gradientId ?? `line-area-fill-${reactId.replace(/:/g, "")}`;
    const areaPath = `${linePath} L 100 100 L 0 100 Z`;

    return (
      <div ref={ref} className={cn("relative", className)} {...props}>
        {/* Y axis labels */}
        <div className="absolute left-0 top-0 bottom-5 flex flex-col justify-between text-[10px] text-s4e-text-disabled">
          {yLabels.map((label) => (
            <span key={label}>{label}</span>
          ))}
        </div>

        {/* Chart */}
        <div className="ml-10">
          <svg viewBox="0 0 100 100" preserveAspectRatio="none" className="w-full h-40">
            <defs>
              <linearGradient id={fillId} x1="0" y1="0" x2="0" y2="1">
                <stop offset="0%"   stopColor="var(--s4e-brand-primary-500)" stopOpacity="0.28" />
                <stop offset="100%" stopColor="var(--s4e-brand-primary-500)" stopOpacity="0"    />
              </linearGradient>
            </defs>
            {[0, 33, 66, 100].map((y) => (
              <line
                key={y}
                x1="0"
                x2="100"
                y1={y}
                y2={y}
                stroke="var(--s4e-neutral-grey-200)"
                strokeWidth="0.3"
                strokeDasharray="1 1"
              />
            ))}
            {showArea && <path d={areaPath} fill={`url(#${fillId})`} />}
            <path
              d={linePath}
              fill="none"
              stroke="var(--s4e-brand-primary-500)"
              strokeWidth="0.9"
              strokeLinecap="round"
              strokeLinejoin="round"
            />
          </svg>
        </div>

        {/* X axis labels */}
        {xLabels.length > 0 && (
          <div className="ml-10 mt-2 flex justify-between text-[10px] text-s4e-text-disabled">
            {xLabels.map((d) => (
              <span key={d} className="whitespace-nowrap">{d}</span>
            ))}
          </div>
        )}
      </div>
    );
  },
);
LineChart.displayName = "LineChart";

// ── Treemap ────────────────────────────────────────────────────────────────

export type TreemapBlock = {
  label:    string;
  value:    number;
  color:    string;
  textOn?:  "accent" | "primary";
};

export interface TreemapChartProps extends React.HTMLAttributes<HTMLDivElement> {
  primary:    TreemapBlock[];
  secondary?: TreemapBlock[];
  height?:    number;
}

export const TreemapChart = React.forwardRef<HTMLDivElement, TreemapChartProps>(
  ({ primary, secondary, height = 256, className, ...props }, ref) => {
    const renderBlock = (b: TreemapBlock) => (
      <div
        key={b.label}
        className={cn(
          "flex-1 rounded-md flex items-center justify-center text-[13px] font-semibold",
          b.textOn === "primary" ? "text-s4e-text-primary" : "text-s4e-text-on-accent",
        )}
        style={{ backgroundColor: b.color }}
      >
        {b.label} · {b.value}
      </div>
    );

    return (
      <div
        ref={ref}
        className={cn("flex gap-1", className)}
        style={{ height }}
        {...props}
      >
        {primary.map(renderBlock)}
        {secondary && secondary.length > 0 && (
          <div className="flex-1 flex flex-col gap-1">
            {secondary.map(renderBlock)}
          </div>
        )}
      </div>
    );
  },
);
TreemapChart.displayName = "TreemapChart";
